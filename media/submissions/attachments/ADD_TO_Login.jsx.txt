const pendingToken = localStorage.getItem("pendingJoinToken");
if (pendingToken) {
  localStorage.removeItem("pendingJoinToken");
  navigate(`/join/${pendingToken}`);
} else {
  navigate("/tournaments"); // або куди редиректиш зараз
}
